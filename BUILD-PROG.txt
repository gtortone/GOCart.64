
- build GOCart.64 firmware (the launcher is built by CMake too, with cc65;
  options: -DCC65_BIN_DIR=<dir> -DCC65_HOME=<dir> -DGOCART_LAUNCHER_VERSION=<x.y>
  -DGOCART_LAUNCHER_THEME=<n>)

cmake -B build/release -DPICO_BOARD=weact_rp2350b_core
cmake -B build/debug -DPICO_BOARD=weact_rp2350b_core -DDEBUG=ON

  with the PSRAM (8 MB, CS rewired to GPIO 47):

cmake -B build/release -DPICO_BOARD=weact_rp2350b_core_psram
cmake -B build/debug -DPICO_BOARD=weact_rp2350b_core_psram -DDEBUG=ON

openocd

~/fun/openocd/src/openocd -s ~/fun/openocd/tcl -f interface/cmsis-dap.cfg -f target/rp2350.cfg -c "adapter speed 20000"

- to program firmware with launcher

program build/debug/GOCart.64.img.bin 0x10000000 verify reset
